async function checkout(){ if(!cart.length){ toast('กรุณาเลือกสินค้าก่อน'); return; } const total = cartTotal(); let received = null; let changeAmount = 0; if(payMethod === 'cash'){ received = Number($('received').value || 0); if(received <= 0){ toast('กรุณาใส่จำนวนเงินที่รับมา'); $('received').focus(); return; } if(received < total){ toast('เงินที่รับมาไม่พอ ต้องรับอย่างน้อย ' + baht(total)); return; } changeAmount = received - total; } const payload = { cashier_id: profile.id, payment_method: payMethod, total: total, received: received, change_amount: changeAmount, status: 'paid' }; console.log('กำลังสร้างออเดอร์:', payload); const { data: order, error: orderError } = await sb .from('orders') .insert(payload) .select('id, order_no, total, payment_method') .single(); if(orderError){ console.error('ORDER ERROR:', orderError); toast('สร้างออเดอร์ไม่ได้: ' + orderError.message); return; } if(!order || !order.id){ toast('สร้างออเดอร์ไม่สำเร็จ: ไม่พบเลขออเดอร์'); return; } const items = cart.map(item => ({ order_id: order.id, product_id: item.id, product_name: item.name, unit_price: Number(item.price), quantity: Number(item.qty) })); console.log('กำลังบันทึกรายการ:', items); const { error: itemError } = await sb .from('order_items') .insert(items); if(itemError){ console.error('ORDER ITEMS ERROR:', itemError); toast('เพิ่มรายการในออเดอร์ไม่ได้: ' + itemError.message); return; } toast( '✅ บันทึกออเดอร์ #' + (order.order_no || order.id) + ' สำเร็จ ' + baht(total) ); cart = []; $('received').value = ''; renderCart(); await loadOrders(); renderOrders(); renderDashboard(); }
